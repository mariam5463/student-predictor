function register() { let username = document.getElementById("regUsername"); let password = document.getElementById("regPassword"); if (!username || !password) { alert("Form error!"); return; } let u = username.value.trim(); let p = password.value.trim(); if (u === "" || p === "") { alert("Fill all fields!"); return; } let users = JSON.parse(localStorage.getItem("users")) || []; let exists = users.find(user => user.username === u); if (exists) { document.getElementById("msg").innerText = "User already exists!"; return; } users.push({ username: u, password: p }); localStorage.setItem("users", JSON.stringify(users)); document.getElementById("msg").innerText = "Registered Successfully!"; }
